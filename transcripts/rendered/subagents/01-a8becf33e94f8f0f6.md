    79	    if status not in MISSION_ALLOWED_FROM[MissionAction.EDIT]:
    80	        raise StateConflict(_wrong_state(status, MissionAction.EDIT.value))
    81	    return _after_change(status)
    82	
    83	
    84	def state_problem(status: MissionStatus, action: MissionAction) -> Problem | None:
    85	    """Why `action` isn't allowed from `status`, or None if it is."""
    86	    if status in MISSION_ALLOWED_FROM[action]:
    87	        return None
    88	    return _wrong_state(status, action.value)
    89	
    90	
    91	def mission_status_after(
    92	    status: MissionStatus, action: MissionAction, *, reaches_quorum: bool = False
    93	) -> MissionStatus:
    94	    """The mission's status once `action` succeeds. Check `state_problem` first."""
    95	    if state_problem(status, action) is not None:
    96	        raise ValueError(f"{action.value} is not allowed from {status.value}")
    97	    match action:
    98	        case MissionAction.SUBMIT:
    99	            return M.PENDING_APPROVAL
   100	        case MissionAction.APPROVE:
   101	            return M.ACTIVE if reaches_quorum else M.PENDING_APPROVAL
   102	        case MissionAction.REJECT:
   103	            return M.DRAFT
   104	        case MissionAction.COMPLETE:
   105	            return M.COMPLETED
   106	        case MissionAction.CANCEL:
   107	            return M.CANCELLED
   108	        case MissionAction.EDIT | MissionAction.TRANSFER:
   109	            return _after_change(status)

   115	class AssignmentAction(StrEnum):
   116	    PROPOSE = "propose"
   117	    ACCEPT = "accept"

   119	    WITHDRAW = "withdraw"
   120	    REMOVE = "remove"

   122	
   123	# action -> (assignment states it may start from, mission states it's allowed in).
   124	# PROPOSE creates the assignment, so it has no starting state.
   125	_ASSIGNMENT_RULES: Mapping[
   126	    AssignmentAction, tuple[frozenset[AssignmentStatus] | None, frozenset[MissionStatus]]
   127	] = MappingProxyType(
   128	    {
   129	        AssignmentAction.PROPOSE: (None, frozenset({M.DRAFT, M.PENDING_APPROVAL})),
   130	        AssignmentAction.ACCEPT: (frozenset({A.PROPOSED}), frozenset({M.DRAFT})),
   131	        AssignmentAction.DECLINE: (frozenset({A.PROPOSED}), frozenset({M.DRAFT})),
   132	        AssignmentAction.WITHDRAW: (
   133	            frozenset({A.ACCEPTED}),
   134	            frozenset({M.DRAFT, M.PENDING_APPROVAL}),
   135	        ),
   136	        AssignmentAction.REMOVE: (
   137	            frozenset({A.PROPOSED, A.ACCEPTED}),
   138	            frozenset({M.DRAFT, M.PENDING_APPROVAL}),
   139	        ),
   140	    }
   141	)
   142	
   143	_ASSIGNMENT_RESULT: Mapping[AssignmentAction, AssignmentStatus] = MappingProxyType(
   144	    {
   145	        AssignmentAction.PROPOSE: A.PROPOSED,
   146	        AssignmentAction.ACCEPT: A.ACCEPTED,
   147	        AssignmentAction.DECLINE: A.DECLINED,
   148	        AssignmentAction.WITHDRAW: A.WITHDRAWN,
   149	        AssignmentAction.REMOVE: A.REMOVED,
   150	    }
   151	)

   153	# How each assignment action reads before "a mission ...".
   154	_ASSIGNMENT_PHRASE: Mapping[AssignmentAction, str] = MappingProxyType(
   155	    {
   156	        AssignmentAction.PROPOSE: "propose crew for",
   157	        AssignmentAction.ACCEPT: "accept a proposal on",
   158	        AssignmentAction.DECLINE: "decline a proposal on",
   159	        AssignmentAction.WITHDRAW: "withdraw from",
   160	        AssignmentAction.REMOVE: "remove crew from",
   161	    }
   162	)
   163	
   164	# Actions that change the roster, and so send a pending mission back to Draft.
   165	_ROSTER_CHANGES = frozenset(
   166	    {AssignmentAction.PROPOSE, AssignmentAction.WITHDRAW, AssignmentAction.REMOVE}
   167	)
   168	
   169	
   170	def assignment_problem(
   171	    action: AssignmentAction, current: AssignmentStatus | None, mission: MissionStatus
   172	) -> Problem | None:
   173	    """Why `action` isn't allowed, or None. Mission state is checked before assignment state."""
   174	    from_states, mission_states = _ASSIGNMENT_RULES[action]
   175	    if mission not in mission_states:
   176	        return _wrong_state(mission, _ASSIGNMENT_PHRASE[action])
   177	    if from_states is not None and current not in from_states:
   178	        label = current.value if current else "missing"
   179	        return Problem("invalid_transition", f"Can't {action.value} an assignment that is {label}")
   180	    return None
   181	
   182	
   183	def assignment_status_after(action: AssignmentAction) -> AssignmentStatus:
   184	    return _ASSIGNMENT_RESULT[action]
   185	
   186	
   187	def mission_status_after_assignment_action(
   188	    mission: MissionStatus, action: AssignmentAction
   189	) -> MissionStatus:
   190	    return _after_change(mission) if action in _ROSTER_CHANGES else mission
   191	
   192	
   193	def cascade(mission_final_status: MissionStatus, assignment: AssignmentStatus) -> AssignmentStatus:
   194	    """What an assignment becomes when its mission completes or is cancelled."""
   195	    if mission_final_status == M.COMPLETED and assignment == A.ACCEPTED:
   196	        return A.COMPLETED
   197	    if mission_final_status == M.CANCELLED and assignment in (A.PROPOSED, A.ACCEPTED):
   198	        return A.CANCELLED
   199	    return assignment
   200	
   201	
   202	# Submit guard
   203	
   204	
   205	def roster_problems(roster: Sequence[RosterEntry], window: DateRange) -> list[Problem]:
   206	    """Accepted crew who are no longer eligible, e.g. after a skill change. Checked at submit
   207	    and again at every approval."""
   208	    return [
   209	        p
   210	        for e in roster
   211	        if e.status == A.ACCEPTED
   212	        for p in ineligibility_reasons(e.crew, e.slot, window)
   213	    ]
   214	
   215	
   216	def check_submit(
   217	    slots: Sequence[SlotSpec],
   218	    roster: Sequence[RosterEntry],
   219	    window: DateRange,
   220	    eligible_approvers: int,
   221	    approvals_required: int,
   222	) -> list[Problem]:
   223	    """Everything stopping a Draft mission being submitted, all at once. `slots[i].open_places`
   224	    is headcount minus open assignments.
   225	
   226	    The order is part of the contract, and callers show problems in it: unfilled places (in
   227	    slot order), then unanswered proposals, then accepted crew who are no longer eligible (in
   228	    roster order), then too few eligible approvers."""
   229	    problems: list[Problem] = []
   230	    for s in slots:
   231	        if s.open_places > 0:
   232	            problems.append(
   233	                Problem(
   234	                    "slot_unfilled",
   235	                    f"{s.name}: {s.open_places} place{'s' if s.open_places > 1 else ''} unfilled",
   236	                    {"slot_id": s.id},
   237	                )
   238	            )
   239	    for e in roster:
   240	        if e.status == A.PROPOSED:
   241	            problems.append(
   242	                Problem(
   243	                    "proposal_unanswered",
   244	                    f"{e.slot.name}: {e.crew.name} hasn't responded to the proposal",
   245	                    {
   246	                        "assignment_id": e.assignment_id,
   247	                        "slot_id": e.slot.id,
   248	                        "crew_profile_id": e.crew.crew_profile_id,
   249	                    },
   250	                )
   251	            )
   252	    problems.extend(roster_problems(roster, window))
   253	    if eligible_approvers < approvals_required:
   254	        problems.append(
   255	            Problem(
   256	                "insufficient_approvers",
   257	                f"{eligible_approvers} eligible approver{'s' if eligible_approvers != 1 else ''}, "
   258	                f"{approvals_required} required",
   259	            )
   260	        )
   261	    return problems
